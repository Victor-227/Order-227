			subtract_from_variable = {

			}

				}

					has_country_flag = SOV_focus_on_progression
				}

				add_to_variable = { SOV_second_five_year_plan_progress = 4 }
			}

				add_to_variable = { SOV_second_five_year_plan_progress = 3 }
			}

				SOV_fyp_consumer_goods_expected_value = -0.01
